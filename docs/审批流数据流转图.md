  │ 新负责人 (operators)│                      │ 新负责人 (外部数据源) │  │
  │ 生效日期 + 原因     │                      │ 生效日期 + 原因     │  │
  └────────┬─────────┘                      └────────┬─────────┘  │
           │                                        │              │
           ▼                                        ▼              │
  POST /api/internal                     飞书原生表单提交            │
  /approval/submit                             │                  │
           │                                    │                  │

  │    widget_old_op    = operator_id (radio → option.key)      │   │
  │    widget_new_op    = operator_id (radio → option.key)      │   │
  │    widget_date      = YYYY-MM-DD                            │   │

  └──────────────────────────┬──────────────────────────────────┘   │
                             │                                      │
                             ▼                                      │

  ║  POST /approval/v4/instances                                 ║   │
  ║    approval_code = 9959B9BB-...                              ║   │
  ║    form = JSON(formValues)      ← ID 全在这里                ║   │
  ║    node_approver_open_id_list   ← 审批人                     ║◄──┘

                              │

  ║                                                              ║

  ║                                                              ║

  ║       SET effective_to = 生效日期 - 1天                        ║
  ║                                                              ║
  ║    2. shop_operators (新负责人)                               ║
  ║       UPSERT is_primary = true                                ║
  ║       SET effective_from = 生效日期                           ║
  ║       SET effective_to = null                                 ║
  ║                                                              ║
  ║    3. shop_operator_changes                                  ║
  ║       INSERT/UPDATE status = 'approved'                      ║
  ║       SET approved_at = NOW()                                ║
  ║                                                              ║
  ║  IF status = REJECTED:                                       ║
  ║    shop_operator_changes.status = 'rejected'                 ║
  ╚══════════════════════════════════════════════════════════════╝

                              ▼
  ╔══════════════════════════════════════════════════════════════╗

  ║    → GMV 100% 归属主负责人                                    ║
  ║                                                              ║
  ║  /internal/admin → 店铺绑定                                   ║
  ║    → 主负责人绿色徽章 + 日期范围                               ║
  ║                                                              ║

  ║    → 变动记录表 (shop_operator_changes)                       ║
