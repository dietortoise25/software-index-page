```
1. 前端跳转飞书授权页 → 用户扫码

```

```typescript
/**
 * 飞书 OAuth 2.0 自定义插件
 */
import type { BetterAuthPlugin } from "better-auth"

interface FeishuOAuthConfig {
  appId: string
  appSecret: string
  redirectUri: string
}

const FEISHU_AUTH_URL = "https://open.feishu.cn/open-apis/authen/v1/authorize"
const FEISHU_TOKEN_URL = "https://open.feishu.cn/open-apis/authen/v1/oidc/access_token"
const FEISHU_USER_INFO_URL = "https://open.feishu.cn/open-apis/authen/v1/user_info"

export function feishuOAuth(config: FeishuOAuthConfig): BetterAuthPlugin {
  return {
    id: "feishu-oauth",
    init(ctx) {
      return {
        async signIn(body: { code: string; redirectUri: string }) {
          // 1. 用 code 换 token
          const tokenRes = await fetch(FEISHU_TOKEN_URL, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({
              grant_type: "authorization_code",
              code: body.code,
              app_id: config.appId,
              app_secret: config.appSecret,
            }),
          })
          const tokenData = await tokenRes.json() as any
          if (!tokenData.data?.access_token) {
            throw new Error("飞书 token 换取失败: " + JSON.stringify(tokenData))
          }

          // 2. 获取用户信息
          const userRes = await fetch(FEISHU_USER_INFO_URL, {
            headers: { Authorization: `Bearer ${tokenData.data.access_token}` },
          })
          const userData = await userRes.json() as any
          if (!userData.data?.name) {
            throw new Error("飞书用户信息获取失败")
          }

          return {
            email: userData.data.email || `${userData.data.open_id}@feishu.user`,
            name: userData.data.name,
            openId: userData.data.open_id,
            avatar: userData.data.avatar_url,
          }
        },

        getAuthorizationUrl(redirectUri: string) {
          const params = new URLSearchParams({
            app_id: config.appId,
            redirect_uri: redirectUri,
            scope: "contact:user.base:readonly",
            state: crypto.randomUUID(),
          })
          return `${FEISHU_AUTH_URL}?${params.toString()}`
        },
      }
    },
  }
}

// 导出配置实例
export const feishuOAuthConfig: FeishuOAuthConfig = {
  appId: process.env.FEISHU_APP_ID || "",
  appSecret: process.env.FEISHU_APP_SECRET || "",
  redirectUri: process.env.FEISHU_REDIRECT_URI || "",
}
```

- [ ] **Step 4: 创建回调路由 server/src/routes/feishu-auth.ts**

```typescript
import { Router } from "express"
import { auth } from "../lib/auth"

const router = Router()

router.get("/callback/feishu", async (req, res) => {
  const { code } = req.query
  if (!code || typeof code !== "string") {
    return res.status(400).send("Missing code")
  }

  try {
    // 调用飞书 OAuth 完成登录
    const result = await auth.api.signInSocial({
      body: {
        provider: "feishu-oauth" as any,
        code,
        callbackURL: `${process.env.AUTH_BASE_URL || "http://localhost:8765"}/api/auth/callback/feishu`,
      },
      headers: new Headers(req.headers as any),
    } as any)

    // 设置 session cookie 并重定向
    if (result?.token) {
      res.cookie("better-auth.session_token", result.token, {
        httpOnly: true,
        secure: false,
        sameSite: "lax",
        maxAge: 7 * 24 * 60 * 60 * 1000,
      })
    }
    res.redirect("/")
  } catch (e) {
    console.error("飞书回调失败:", e)
    res.status(500).send("登录失败")
  }
})

export default router
```
